- 17,988.08 + 8,010.66 = 25,998.74.
- 25,998.74 − 3,694.87 − 9,411.20 − 544.68 = 12,347.99.
- 12,347.99 − 3,222.74 = 9,125.25.
- 9,101.18 + 24.07 = 9,125.25.
